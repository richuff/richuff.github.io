4.完成项目

package main

import (
  "fmt"
  "github.com/gin-gonic/gin"
  "github.com/jinzhu/gorm"
  _ "github.com/jinzhu/gorm/dialects/mysql"
  "net/http"
)

var (
  open *gorm.DB
)

type TodoList struct {
  ID     int    `json:"id"`
  Title  string `json:"title"`
  Status bool   `json:"status"`
}

func initMysql() (err error) {
  //连接数据库
  open, err = gorm.Open("mysql", "root:password@(127.0.0.1:3306)/mp?charset=utf8&parseTime=True&loc=Local")
  if err != nil {
    fmt.Println("error")
    return err
  }
  return open.DB().Ping()
}

func main() {
  r := gin.Default()

  err := initMysql()
  if err != nil {
    return
  }
  defer open.Close()
  //模型绑定
  open.AutoMigrate(&TodoList{})

  r.LoadHTMLGlob("templates/*")
  //告诉gin框架引用的静态文件在哪里
  r.StaticFS("/js", http.Dir("./static/js"))
  r.StaticFS("/css", http.Dir("./static/css"))
  r.StaticFS("/static/fonts", http.Dir("./static/fonts"))
  r.GET("/", func(c *gin.Context) {
    c.HTML(http.StatusOK, "index.html", nil)
  })

  v1 := r.Group("/v1")
  {
    //待办事项
    //添加
    v1.POST("/todo", func(context *gin.Context) {
      //前端页面填写事项，发起请求
      var todo TodoList
      err := context.BindJSON(&todo)
      if err != nil {
        context.JSON(http.StatusOK, gin.H{
          "error": err,
        })
      }
      if err = open.Create(&todo).Error; err != nil {
        context.JSON(http.StatusOK, gin.H{
          "error": err,
        })
      } else {
        context.JSON(http.StatusOK, todo)
      }

    })
    //查看所有
    v1.GET("/todo", func(context *gin.Context) {
      var todoList []TodoList
      if err = open.Find(&todoList).Error; err != nil {
        context.JSON(http.StatusOK, gin.H{
          "error": err.Error(),
        })
      } else {
        context.JSON(http.StatusOK, todoList)
      }
    })
    //查看一个
    v1.GET("/todo/:id", func(context *gin.Context) {

    })
    //修改
    v1.PUT("/todo/:id", func(context *gin.Context) {
      id := context.Param("id")
      var todo TodoList
      if err = open.Where("id = ?", id).First(&todo).Error; err != nil {
        context.JSON(http.StatusOK, gin.H{
          "error": err.Error(),
        })
      } else {
        todo.Status = !todo.Status
        if err = open.Save(&todo).Error; err != nil {
          context.JSON(http.StatusOK, gin.H{
            "error": err.Error(),
          })
        } else {
          context.JSON(http.StatusOK, todo)
        }
      }
    })
    //删除
    v1.DELETE("/todo/:id", func(context *gin.Context) {
      id := context.Param("id")
      var todo TodoList
      if err = open.Where("id = ?", id).First(&todo).Error; err != nil {
        context.JSON(http.StatusOK, gin.H{
          "error": err.Error(),
        })
      } else {
        if err = open.Delete(&todo).Error; err != nil {
          context.JSON(http.StatusOK, gin.H{
            "error": err.Error(),
          })
        } else {
          context.JSON(http.StatusOK, todo)
        }
      }
    })
  }

  err = r.Run(":8080")
  if err != nil {
    fmt.Println(err)
  }
}

4.md5加密

package utils

import (
  "crypto/md5"
  "encoding/hex"
  "strings"
)

// 小写
func Md5Encode(data string) string {
  h := md5.New()
  h.Write([]byte(data))
  tempStr := h.Sum(nil)
  return hex.EncodeToString(tempStr)
}

// 大写
func MD5Encode(data string) string {
  return strings.ToUpper(Md5Encode(data))
}

// 加密
func MakePassword(plainpwd string, salt string) string {
  return Md5Encode(plainpwd + salt)
}

// 解密
func ValidPassword(piainpwd string, salt string) string {
  return Md5Encode(piainpwd + salt)
}

5.go使用redis

go get github.com/redis/go-redis/v9

import (
  "context"
  "fmt"
  "github.com/redis/go-redis/v9"
)

var (
  Red   *redis.Client
)

func InitRedis(addr string, password string, db int, poolSize int, minIdleConns int) (string, error) {
  Red = redis.NewClient(&redis.Options{
    Addr:         addr,         // Redis 服务器地址和端口
    Password:     password,     // Redis 访问密码，如果没有可以为空字符串
    DB:           db,           // 使用的 Redis 数据库编号，默认为 0
    PoolSize:     poolSize,     //指定连接池中的最大连接数
    MinIdleConns: minIdleConns, //指定连接池中的最小空闲连接数
  })
  pong, err := Red.Ping(context.Background()).Result()
  if err != nil {
    fmt.Println(err)
  } else {
    fmt.Println(pong)
  }
  return pong, err
}

6.go使用websocket

go get github.com/gorilla/websocket

gorilla/websocket 是一个广泛使用且经过良好测试的 WebSocket 实现，适合大多数 WebSocket 应用开发。其主要特点包括：

简单易用：提供直观的 API，如 WriteMessage 和 ReadMessage。支持多种消息类型：包括文本消息、二进制消息、Ping/Pong 消息等。并发支持：一个连接支持一个并发 reader 和一个并发 writer。

package main

import (
    "github.com/gorilla/websocket"
    "log"
    "net/http"
)

var upgrader = websocket.Upgrader{
    ReadBufferSize: 1024,
    WriteBufferSize: 1024,
    CheckOrigin: func(r *http.Request) bool {
    return true
    },
}

func handler(w http.ResponseWriter, r *http.Request) {
    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil {
    log.Fatal(err)
    }
    conn.WriteMessage(websocket.TextMessage, []byte("Hello, World!"))
    conn.Close()
}

func main() {
    http.HandleFunc("/ws", handler)
    http.ListenAndServe(":8181", nil)
}
