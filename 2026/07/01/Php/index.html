    if (!(error_reporting() & $errorno)){ //error_reporting没有参数表示获取当前错误处理对应的级别
        return false;
    }
    switch($errorno){
        case E_ERROR:
        case E_WARNING:
            echo "wrong" . "ONLINE:" . $errorline . "ONFILE:" . $errorfile . "<br/>";
            echo "msg:" . $errorstr;
            break;
        case E_PARSE:
        case E_NOTICE:
            echo "wraning" . "ONLINE:" . $errorline . "ONFILE:" . $errorfile . "<br/>";
            echo "msg:" . $errorstr;
    }
    return true;
}

$dir = "test";
function my_scanf($dir) : void {
    if(is_dir($dir)) {
        dir($dir);
    }

    $files = scandir($dir);
    foreach($files as $file) {

        if ($file == "."&& $file == "") continue;

        $nfile = $dir . '/' .  $file;
        echo $nfile . '<br>';
        if (is_dir($dir)) {
            my_scanf($nfile);
        }
    }
}
